cmake_minimum_required(VERSION 3.28)
project(pico_wifi_lvgl_test C)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${APP_ROOT}/modules/lib/gui/lvgl" lvgl)
add_executable(wifi_lvgl_test test.c "${APP_ROOT}/src/wifi_ui_lvgl.c" "${APP_ROOT}/src/display_font.c" "${APP_ROOT}/src/dashboard_lvgl.c" "${APP_ROOT}/src/ui_cards.c")
target_compile_definitions(wifi_lvgl_test PRIVATE CONFIG_PICO_SENSE_LVGL_WIFI=1)
target_include_directories(wifi_lvgl_test PRIVATE stubs "${APP_ROOT}/src")
target_link_libraries(wifi_lvgl_test PRIVATE lvgl)
enable_testing()
add_test(NAME wifi_lvgl_test COMMAND wifi_lvgl_test)
set_tests_properties(wifi_lvgl_test PROPERTIES TIMEOUT 20)
